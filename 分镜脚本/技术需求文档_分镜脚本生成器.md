### 7.1 Word文档生成

```python
from docx import Document
from docx.shared import Pt, Cm, Inches, RGBColor

class StoryboardDocumentGenerator:
    """分镜脚本Word文档生成器"""
    
    def __init__(self, content: dict, project_info: dict):
        self.content = content
        self.project_info = project_info
        self.doc = Document()
    
    def generate(self) -> str:
        """生成Word文档，返回文件路径"""
        self._setup_styles()
        self._add_cover()
        self._add_toc()
        self._add_content()
        
        output_path = self._get_output_path()
        self.doc.save(output_path)
        return output_path
    
    def _setup_styles(self):
        """设置文档样式"""
        # 设置页边距
        for section in self.doc.sections:
            section.top_margin = Cm(2.54)
            section.bottom_margin = Cm(2.54)
            section.left_margin = Cm(3.17)
            section.right_margin = Cm(3.17)
    
    def _add_cover(self):
        """添加封面"""
        # 标题
        title = self.doc.add_heading(self.project_info['name'], 0)
        title.alignment = WD_ALIGN_PARAGRAPH.CENTER
        
        # 版本信息
        version_para = self.doc.add_paragraph()
        version_para.alignment = WD_ALIGN_PARAGRAPH.CENTER
        run = version_para.add_run(f"【{self.content['version_type']}】")
        
        self.doc.add_page_break()
    
    def _add_toc(self):
        """添加目录"""
        self.doc.add_heading('目录', level=1)
        # ... 生成目录
        self.doc.add_page_break()
    
    def _add_content(self):
        """添加正文内容"""
        for section in self.content['sections']:
            self._add_section(section)
    
    def _add_section(self, section: dict):
        """添加单个章节"""
        section_type = section['type']
        content = section['content']
        
        if section_type == 'heading1':
            self.doc.add_heading(content, level=1)
        elif section_type == 'heading2':
            self.doc.add_heading(content, level=2)
        elif section_type == 'paragraph':
            self.doc.add_paragraph(content)
        elif section_type == 'quote':
            para = self.doc.add_paragraph()
            para.paragraph_format.left_indent = Cm(1)
            run = para.add_run(content)
            run.italic = True
        elif section_type == 'table':
            self._add_table(content)
    
    def _add_table(self, table_data: list):
        """添加表格"""
        if not table_data:
            return
        
        rows = len(table_data)
        cols = len(table_data[0])
        table = self.doc.add_table(rows=rows, cols=cols)
        
        for i, row_data in enumerate(table_data):
            row = table.rows[i]
            for j, cell_text in enumerate(row_data):
                row.cells[j].text = str(cell_text)
        
        # 应用表格样式
        self._style_table(table)
```

## 八、异步任务处理

### 8.1 Celery任务定义

```python
from celery import Celery
from celery.result import AsyncResult

app = Celery('storyboard',
             broker='redis://localhost:6379/0',
             backend='redis://localhost:6379/1')

@app.task(bind=True)
def generate_storyboard_task(self, project_id: str, mode: str, **kwargs):
    """异步生成分镜脚本任务"""
    try:
        # 更新任务状态
        self.update_state(state='PROCESSING', meta={'progress': 0})
        
        # 1. 获取项目信息
        project = get_project(project_id)
        self.update_state(state='PROCESSING', meta={'progress': 20})
        
        # 2. 组装Prompt
        prompt = build_prompt(project, mode)
        self.update_state(state='PROCESSING', meta={'progress': 30})
        
        # 3. 调用LLM生成
        raw_content = call_llm(prompt)
        self.update_state(state='PROCESSING', meta={'progress': 70})
        
        # 4. 解析内容
        parsed_content = parse_content(raw_content)
        self.update_state(state='PROCESSING', meta={'progress': 85})
        
        # 5. 保存版本
        version = save_version(project_id, parsed_content, mode)
        self.update_state(state='PROCESSING', meta={'progress': 95})
        
        # 6. 生成文档
        generate_documents(version.id)
        
        return {
            'status': 'completed',
            'version_id': str(version.id),
            'progress': 100
        }
        
    except Exception as e:
        self.update_state(state='FAILED', meta={'error': str(e)})
        raise
```

```python
async def get_generation_status(task_id: str):
    """查询生成任务状态"""
    task = AsyncResult(task_id)
    
    if task.ready():
        if task.successful():
            return {
                "status": "completed",
                "result": task.result
            }
        else:
            return {
                "status": "failed",
                "error": str(task.result)
            }
    else:
        return {
            "status": task.state.lower(),
            "progress": task.info.get('progress', 0)
        }
```
